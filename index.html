local viewport = Instance.new("ViewportFrame")
viewport.Size = UDim2.fromOffset(300, 300)
viewport.BackgroundTransparency = 1
viewport.Ambient = Color3.fromRGB(90, 90, 90)
viewport.LightColor = Color3.fromRGB(255, 255, 255)
viewport.LightDirection = Vector3.new(-1, -1, -1)
viewport.Parent = screenGui

-- 2) 
local model = game.ReplicatedStorage.Sword:Clone()
model.Parent = viewport

-- 3) 
local camera = Instance.new("Camera")
camera.FieldOfView = 45
camera.CFrame = CFrame.new(Vector3.new(0, 0, 5), Vector3.zero)
camera.Parent = viewport
viewport.CurrentCamera = camera

-- 4) 
local angle = 0
game:GetService("RunService").RenderStepped:Connect(function(dt)
    angle += dt
    model:PivotTo(CFrame.Angles(0, angle, 0))
end)
